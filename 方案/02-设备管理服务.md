```python
async def _run(self, *args: str, timeout: int | None = None) -> str:
    try:
        proc = await asyncio.create_subprocess_exec(...)
        stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=timeout)
        
        if proc.returncode != 0:
            error_msg = stderr.decode().strip()
            raise AdbError(f"ADB command failed: {error_msg}")
        
        return stdout.decode().strip()
    except asyncio.TimeoutError:
        raise AdbError(f"ADB command timeout after {timeout}s")
    except FileNotFoundError:
        raise AdbError("ADB not found. Please install Android SDK Platform Tools")
```

**方案**：

```python
class DeviceService:
    def __init__(self, adb: AdbDriver, repo: DeviceRepository):
        self.adb = adb
        self.repo = repo
        self._refresh_task: asyncio.Task | None = None
    
    async def start(self):
        """Start background device refresh"""
        self._refresh_task = asyncio.create_task(self._refresh_loop())
    
    async def _refresh_loop(self):
        """Refresh device list every 5 seconds"""
        while True:
            try:
                await self.list_devices()
            except Exception as e:
                logger.error("device_refresh_failed", error=str(e))
            await asyncio.sleep(5)
    
    async def stop(self):
        """Stop background refresh"""
        if self._refresh_task:
            self._refresh_task.cancel()
```

#### 2.2 设备事件通知

**问题**：设备上下线无法及时感知

**方案**：
```python
from typing import Callable, Awaitable

class DeviceService:
    def __init__(self, ...):
        self._on_device_connected: list[Callable[[DeviceInfo], Awaitable[None]]] = []
        self._on_device_disconnected: list[Callable[[str], Awaitable[None]]] = []
    
    def on_device_connected(self, callback: Callable[[DeviceInfo], Awaitable[None]]):
        """Register device connected callback"""
        self._on_device_connected.append(callback)
    
    def on_device_disconnected(self, callback: Callable[[str], Awaitable[None]]):
        """Register device disconnected callback"""
        self._on_device_disconnected.append(callback)
    
    async def _refresh_loop(self):
        previous_devices = set()
        while True:
            current_devices = set(await self.adb.list_devices())
            
            # New devices
            for device_id in current_devices - previous_devices:
                info = await self.adb.get_device_info(device_id)
                await self.repo.save(info)
                for callback in self._on_device_connected:
                    await callback(info)
            
            # Removed devices
            for device_id in previous_devices - current_devices:
                await self.repo.delete(device_id)
                for callback in self._on_device_disconnected:
                    await callback(device_id)
            
            previous_devices = current_devices
            await asyncio.sleep(5)
```

#### 2.3 批量操作优化

**问题**：批量安装时一个失败影响全部

**方案**：
```python
async def batch_install(self, device_ids: list[str], apk_path: str) -> dict[str, bool]:
    """Batch install with individual error handling"""
    results = {}
    
    async def install_one(device_id: str):
        try:
            await self.adb.install(device_id, apk_path)
            results[device_id] = True
        except Exception as e:
            logger.error("batch_install_failed", device=device_id, error=str(e))
            results[device_id] = False
    
    # 并发执行，限制并发数
    semaphore = asyncio.Semaphore(5)
    
    async def install_with_limit(device_id: str):
        async with semaphore:
            await install_one(device_id)
    
    await asyncio.gather(*[install_with_limit(d) for d in device_ids])
    return results
```

#### 3.2 设备事件流（SSE）

```python
from fastapi.responses import StreamingResponse

@router.get("/events")
async def device_events(service: DeviceService = Depends(get_device_service)):
    """Server-Sent Events for device changes"""
    async def event_stream():
        queue = asyncio.Queue()
        
        async def on_connected(device: DeviceInfo):
            await queue.put({"type": "connected", "device": device})
        
        async def on_disconnected(device_id: str):
            await queue.put({"type": "disconnected", "device_id": device_id})
        
        service.on_device_connected(on_connected)
        service.on_device_disconnected(on_disconnected)
        
        try:
            while True:
                event = await queue.get()
                yield f"data: {json.dumps(event)}\n\n"
        finally:
            service._on_device_connected.remove(on_connected)
            service._on_device_disconnected.remove(on_disconnected)
    
    return StreamingResponse(event_stream(), media_type="text/event-stream")
```

## 技术细节

### ADB 命令参考

```bash
# 列出设备
adb devices

# 设备信息
adb -s <device> shell getprop ro.product.model
adb -s <device> shell getprop ro.build.version.release
adb -s <device> shell wm size
adb -s <device> shell dumpsys battery

# Shell 命令
adb -s <device> shell <command>

# 文件传输
adb -s <device> push <local> <remote>
adb -s <device> pull <remote> <local>

# 安装
adb -s <device> install -r <apk>

# 截图
adb -s <device> shell screencap -p /sdcard/screenshot.png
adb -s <device> pull /sdcard/screenshot.png

# TCP/IP 连接
adb connect <ip>:<port>
adb disconnect <ip>:<port>
```

```python
@dataclass
class DeviceInfo:
    id: str                    # 设备序列号
    model: str                 # 设备型号
    os_version: str            # Android 版本
    resolution: tuple[int, int] # 分辨率 (宽, 高)
    battery: int               # 电量百分比
    status: str                # online / offline / busy
    ip: Optional[str]          # IP 地址（TCP/IP 连接）
    port: Optional[int]        # 端口（TCP/IP 连接）
```
