## 后端实现

### Step 1: 性能数据采集 ⏳

**文件**：`backend/app/infrastructure/performance/sampler.py`

```python
import asyncio
import re
from dataclasses import dataclass
from typing import AsyncIterator

from app.core.logging import get_logger
from app.domain.ports import AdbDriver

logger = get_logger(__name__)

@dataclass
class PerformanceMetrics:
    ts: float
    cpu_percent: float
    total_memory_mb: float
    used_memory_mb: float
    fps: float | None
    jank_frames: int
    current_activity: str

class PerformanceSampler:
    """Collect performance metrics from device"""
    
    def __init__(self, adb: AdbDriver, device_id: str, interval: float = 1.0):
        self.adb = adb
        self.device_id = device_id
        self.interval = interval
        self._running = False
    
    async def start(self) -> AsyncIterator[PerformanceMetrics]:
        """Start sampling"""
        self._running = True
        
        while self._running:
            try:
                metrics = await self._collect()
                yield metrics
            except Exception as e:
                logger.error("performance_sampling_failed", error=str(e))
            
            await asyncio.sleep(self.interval)
    
    async def stop(self):
        """Stop sampling"""
        self._running = False
    
    async def _collect(self) -> PerformanceMetrics:
        """Collect all metrics"""
        import time
        
        cpu = await self._get_cpu_usage()
        mem = await self._get_memory_usage()
        fps = await self._get_fps()
        activity = await self._get_current_activity()
        
        return PerformanceMetrics(
            ts=time.time(),
            cpu_percent=cpu,
            total_memory_mb=mem["total"],
            used_memory_mb=mem["used"],
            fps=fps,
            jank_frames=0,  # TODO: 计算卡顿帧
            current_activity=activity,
        )
    
    async def _get_cpu_usage(self) -> float:
        """Get CPU usage percentage"""
        output = await self.adb.shell(self.device_id, "top -n 1")
        
        # 解析 CPU 使用率
        # 示例: "CPU: 15% user, 10% system, 75% idle"
        match = re.search(r'(\d+)%\s+idle', output)
        if match:
            idle = int(match.group(1))
            return 100 - idle
        
        return 0.0
    
    async def _get_memory_usage(self) -> dict:
        """Get memory usage in MB"""
        output = await self.adb.shell(self.device_id, "cat /proc/meminfo")
        
        mem_info = {}
        for line in output.split('\n'):
            if 'MemTotal' in line:
                mem_info['total'] = int(line.split()[1]) // 1024  # kB -> MB
            elif 'MemAvailable' in line:
                mem_info['available'] = int(line.split()[1]) // 1024
        
        mem_info['used'] = mem_info.get('total', 0) - mem_info.get('available', 0)
        return mem_info
    
    async def _get_fps(self) -> float | None:
        """Get current FPS"""
        output = await self.adb.shell(
            self.device_id,
            "dumpsys gfxinfo $(pidof com.android.launcher3) | grep 'Total frames rendered'"
        )
        
        match = re.search(r'Total frames rendered:\s*(\d+)', output)
        if match:
            # 简化：返回总帧数（实际需要计算帧率）
            return float(match.group(1))
        
        return None
    
    async def _get_current_activity(self) -> str:
        """Get current foreground activity"""
        output = await self.adb.shell(
            self.device_id,
            "dumpsys activity activities | grep mResumedActivity"
        )
        
        # 示例: "mResumedActivity: ActivityRecord{1234 u0 com.example/.MainActivity}"
        match = re.search(r'Activity.*?(\S+/\S+)', output)
        if match:
            return match.group(1)
        
        return "unknown"
```

### Step 2: 性能服务 ⏳

**文件**：`backend/app/application/performance_service.py`

```python
import asyncio
from typing import AsyncIterator

from app.core.logging import get_logger
from app.domain.ports import AdbDriver
from app.infrastructure.performance.sampler import PerformanceSampler, PerformanceMetrics

logger = get_logger(__name__)

class PerformanceService:
    """Performance monitoring service"""
    
    def __init__(self, adb: AdbDriver):
        self.adb = adb
        self.samplers: dict[str, PerformanceSampler] = {}
        self.metrics_buffer: dict[str, list[PerformanceMetrics]] = {}
    
    async def start_monitoring(self, device_id: str, interval: float = 1.0):
        """Start monitoring device"""
        if device_id in self.samplers:
            return
        
        sampler = PerformanceSampler(self.adb, device_id, interval)
        self.samplers[device_id] = sampler
        self.metrics_buffer[device_id] = []
        
        # 启动采样任务
        asyncio.create_task(self._sampling_loop(device_id, sampler))
    
    async def _sampling_loop(self, device_id: str, sampler: PerformanceSampler):
        """Sampling loop"""
        try:
            async for metrics in sampler.start():
                self.metrics_buffer[device_id].append(metrics)
                
                # 保留最近 1 小时数据（3600 条 @ 1s）
                if len(self.metrics_buffer[device_id]) > 3600:
                    self.metrics_buffer[device_id].pop(0)
        except asyncio.CancelledError:
            logger.info("performance_sampling_stopped", device=device_id)
    
    async def stop_monitoring(self, device_id: str):
        """Stop monitoring device"""
        sampler = self.samplers.pop(device_id, None)
        if sampler:
            await sampler.stop()
        self.metrics_buffer.pop(device_id, None)
    
    async def get_metrics(self, device_id: str, limit: int = 100) -> list[dict]:
        """Get recent metrics"""
        buffer = self.metrics_buffer.get(device_id, [])
        return [m.__dict__ for m in buffer[-limit:]]
    
    async def stream_metrics(self, device_id: str) -> AsyncIterator[PerformanceMetrics]:
        """Stream real-time metrics"""
        last_idx = 0
        
        while True:
            buffer = self.metrics_buffer.get(device_id, [])
            
            if last_idx < len(buffer):
                for metrics in buffer[last_idx:]:
                    yield metrics
                last_idx = len(buffer)
            
            await asyncio.sleep(1.0)
```

### Step 3: HTTP/WS 端点 ⏳

```python
# HTTP 端点
@router.get("/api/performance/{device_id}/metrics")
async def get_metrics(
    device_id: str,
    limit: int = 100,
    service: PerformanceService = Depends(get_performance_service),
):
    """Get performance metrics"""
    metrics = await service.get_metrics(device_id, limit)
    return {"metrics": metrics}

# WebSocket 端点
@router.websocket("/ws/performance/{device_id}")
async def stream_metrics(websocket: WebSocket, device_id: str):
    """Stream real-time metrics"""
    await websocket.accept()
    
    service = get_performance_service()
    await service.start_monitoring(device_id)
    
    try:
        async for metrics in service.stream_metrics(device_id):
            await websocket.send_json(metrics.__dict__)
    except WebSocketDisconnect:
        await service.stop_monitoring(device_id)
```

<script setup lang="ts">
import { ref, computed, onMounted, onUnmounted } from 'vue'
import VChart from 'vue-echarts'
import { use } from 'echarts/core'
import { CanvasRenderer } from 'echarts/renderers'
import { LineChart } from 'echarts/charts'

use([CanvasRenderer, LineChart, GridComponent, TooltipComponent])

const props = defineProps<{ deviceId: string }>()

const metricsHistory = ref<any[]>([])
let ws: WebSocket | null = null

const currentMetrics = computed(() => {
  return metricsHistory.value[metricsHistory.value.length - 1] || {}
})

const cpuChartOption = computed(() => ({
  tooltip: { trigger: 'axis' },

    type: 'category',
    data: metricsHistory.value.map((_, i) => i),

    type: 'value',

    areaStyle: { opacity: 0.3 },
  }],
}))

const memChartOption = computed(() => ({
  tooltip: { trigger: 'axis' },

    type: 'value',

    data: metricsHistory.value.map(m => m.used_memory_mb),
    type: 'line',
    smooth: true,
    areaStyle: { opacity: 0.3 },
  }],
}))

onMounted(() => {
  // 连接 WebSocket
  ws = new WebSocket(`ws://localhost:8000/ws/performance/${props.deviceId}`)

    // 保留最近 60 秒

      metricsHistory.value.shift()
    }
  }
})

onUnmounted(() => {
  ws?.close()
})
