## 实现步骤

### Step 1: WebCodecs 解码器 ⏳

**文件**：`frontend/src/services/videoDecoder.ts`

```typescript
export class VideoDecoder {
  private decoder: VideoDecoder | null = null
  private canvas: HTMLCanvasElement | OffscreenCanvas | null = null
  private ctx: CanvasRenderingContext2D | OffscreenCanvasRenderingContext2D | null = null
  
  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas
    this.ctx = canvas.getContext('2d')
  }
  
  async init() {
    // 检查 WebCodecs 支持
    if (!('VideoDecoder' in window)) {
      throw new Error('WebCodecs not supported')
    }
    
    this.decoder = new VideoDecoder({
      output: (frame: VideoFrame) => {
        this.render(frame)
        frame.close()
      },
      error: (e: Error) => {
        console.error('Decoder error:', e)
      },
    })
    
    // 配置解码器
    await this.decoder.configure({
      codec: 'avc1.42E01E',  // H.264 Baseline
      codedWidth: 1080,
      codedHeight: 1920,
    })
  }
  
  decode(data: Uint8Array) {
    if (!this.decoder) return
    
    // 解析 NALU
    const naluType = data[4] & 0x1F
    
    // 判断是否为关键帧
    const isKeyFrame = naluType === 5  // IDR
    
    const chunk = new EncodedVideoChunk({
      type: isKeyFrame ? 'key' : 'delta',
      timestamp: performance.now() * 1000,  // 微秒
      data: data,
    })
    
    this.decoder.decode(chunk)
  }
  
  private render(frame: VideoFrame) {
    if (!this.ctx || !this.canvas) return
    
    // 绘制帧
    this.ctx.drawImage(frame, 0, 0, this.canvas.width, this.canvas.height)
  }
  
  close() {
    this.decoder?.close()
    this.decoder = null
  }
}
```

### Step 2: 视频播放组件 ⏳

**文件**：`frontend/src/components/stream/VideoPlayer.vue`

```vue
<template>
  <div class="video-player" ref="containerRef">
    <canvas
      ref="canvasRef"
      @mousedown="onMouseDown"
      @mousemove="onMouseMove"
      @mouseup="onMouseUp"
      @touchstart="onTouchStart"
      @touchmove="onTouchMove"
      @touchend="onTouchEnd"
    ></canvas>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onUnmounted } from 'vue'
import { VideoDecoder } from '@/services/videoDecoder'
import { WebSocketService } from '@/services/websocket'

const props = defineProps<{ deviceId: string }>()

const containerRef = ref<HTMLElement>()
const canvasRef = ref<HTMLCanvasElement>()
const decoder = ref<VideoDecoder | null>(null)
const ws = ref<WebSocketService | null>(null)

// 触摸状态
const touchStart = ref<{ x: number; y: number } | null>(null)
const touchStartTime = ref<number>(0)

onMounted(async () => {
  if (!canvasRef.value) return
  
  // 设置 canvas 尺寸
  const rect = containerRef.value!.getBoundingClientRect()
  canvasRef.value.width = rect.width
  canvasRef.value.height = rect.height
  
  // 初始化解码器
  decoder.value = new VideoDecoder(canvasRef.value)
  await decoder.value.init()
  
  // 连接 WebSocket
  ws.value = new WebSocketService(`ws://localhost:8000/ws/video/${props.deviceId}`)
  ws.value.setMessageHandler((data) => {
    if (data instanceof Blob) {
      data.arrayBuffer().then((buffer) => {
        decoder.value?.decode(new Uint8Array(buffer))
      })
    }
  })
  ws.value.connect()
})

onUnmounted(() => {
  decoder.value?.close()
  ws.value?.close()
})

// 坐标映射
function mapCoordinates(clientX: number, clientY: number): { x: number; y: number } {
  if (!canvasRef.value) return { x: 0, y: 0 }
  
  const rect = canvasRef.value.getBoundingClientRect()
  const scaleX = canvasRef.value.width / rect.width
  const scaleY = canvasRef.value.height / rect.height
  
  return {
    x: (clientX - rect.left) * scaleX,
    y: (clientY - rect.top) * scaleY,
  }
}

// 鼠标事件
function onMouseDown(e: MouseEvent) {
  const { x, y } = mapCoordinates(e.clientX, e.clientY)
  touchStart.value = { x, y }
  touchStartTime.value = Date.now()
}

function onMouseMove(e: MouseEvent) {
  if (!touchStart.value) return
  
  const { x, y } = mapCoordinates(e.clientX, e.clientY)
  ws.value?.send({
    action: 'swipe',
    x1: touchStart.value.x,
    y1: touchStart.value.y,
    x2: x,
    y2: y,
    duration: 100,
  })
  
  touchStart.value = { x, y }
}

function onMouseUp(e: MouseEvent) {
  if (!touchStart.value) return
  
  const duration = Date.now() - touchStartTime.value
  
  if (duration < 200) {
    // 点击
    const { x, y } = mapCoordinates(e.clientX, e.clientY)
    ws.value?.send({
      action: 'touch',
      x,
      y,
    })
  }
  
  touchStart.value = null
}

// 触摸事件
function onTouchStart(e: TouchEvent) {
  e.preventDefault()
  const touch = e.touches[0]
  const { x, y } = mapCoordinates(touch.clientX, touch.clientY)
  touchStart.value = { x, y }
  touchStartTime.value = Date.now()
}

function onTouchMove(e: TouchEvent) {
  e.preventDefault()
  if (!touchStart.value) return
  
  const touch = e.touches[0]
  const { x, y } = mapCoordinates(touch.clientX, touch.clientY)
  
  ws.value?.send({
    action: 'swipe',
    x1: touchStart.value.x,
    y1: touchStart.value.y,
    x2: x,
    y2: y,
    duration: 100,
  })
  
  touchStart.value = { x, y }
}

function onTouchEnd(e: TouchEvent) {
  e.preventDefault()
  if (!touchStart.value) return
  
  const duration = Date.now() - touchStartTime.value
  
  if (duration < 200) {
    const touch = e.changedTouches[0]
    const { x, y } = mapCoordinates(touch.clientX, touch.clientY)
    ws.value?.send({
      action: 'touch',
      x,
      y,
    })
  }
  
  touchStart.value = null
}
</script>
```

### Step 3: Web Worker 解码（可选优化）⏳

**问题**：主线程解码可能影响 UI 性能

**方案**：使用 Web Worker + OffscreenCanvas

**文件**：`frontend/src/workers/videoDecoder.worker.ts`

```typescript
import { VideoDecoder as Decoder } from 'videoDecoder'

let decoder: Decoder | null = null
let canvas: OffscreenCanvas | null = null
let ctx: OffscreenCanvasRenderingContext2D | null = null

self.onmessage = async (e) => {
  const { type, data } = e.data
  
  if (type === 'init') {
    canvas = data.canvas as OffscreenCanvas
    ctx = canvas!.getContext('2d')
    
    decoder = new Decoder(canvas!)
    await decoder.init()
  } else if (type === 'decode') {
    decoder?.decode(data)
  } else if (type === 'close') {
    decoder?.close()
  }
}
```

**使用**：

```typescript
// 在 VideoPlayer.vue 中
const worker = new Worker(
  new URL('@/workers/videoDecoder.worker.ts', import.meta.url),
  { type: 'module' }
)

// 初始化
const offscreen = canvasRef.value.transferControlToOffscreen()
worker.postMessage({ type: 'init', canvas: offscreen }, [offscreen])

// 解码
worker.postMessage({ type: 'decode', data: frameData })
```

### Step 4: 手势识别 ⏳

**增强手势支持**：

```typescript
// 双指缩放
function onTouchStart(e: TouchEvent) {
  if (e.touches.length === 2) {
    // 记录初始距离
    const dx = e.touches[0].clientX - e.touches[1].clientX
    const dy = e.touches[0].clientY - e.touches[1].clientY
    initialDistance.value = Math.sqrt(dx * dx + dy * dy)
  }
}

function onTouchMove(e: TouchEvent) {
  if (e.touches.length === 2 && initialDistance.value > 0) {
    const dx = e.touches[0].clientX - e.touches[1].clientX
    const dy = e.touches[0].clientY - e.touches[1].clientY
    const currentDistance = Math.sqrt(dx * dx + dy * dy)
    
    const scale = currentDistance / initialDistance.value
    
    // 发送缩放手势
    ws.value?.send({
      action: 'pinch',
      scale,
    })
  }
}

// 长按
let longPressTimer: number | null = null

function onMouseDown(e: MouseEvent) {
  longPressTimer = window.setTimeout(() => {
    // 长按事件
    const { x, y } = mapCoordinates(e.clientX, e.clientY)
    ws.value?.send({
      action: 'long_press',
      x,
      y,
    })
  }, 500)
}

function onMouseUp() {
  if (longPressTimer) {
    clearTimeout(longPressTimer)
    longPressTimer = null
  }
}
```

### H.264 Codec 字符串

```
avc1.42E01E
│   │││││└── Level (1E = 3.0)
│   │││└── Profile (E = Baseline)
│   │└── Constraint flags
│   └── Profile IDC (42 = 66 = Baseline)
```

## 性能优化

### 1. 帧率限制

```typescript
let lastFrameTime = 0
const MIN_FRAME_INTERVAL = 16  // 60 FPS

function render(frame: VideoFrame) {
  const now = performance.now()
  if (now - lastFrameTime < MIN_FRAME_INTERVAL) {
    frame.close()
    return
  }
  
  lastFrameTime = now
  ctx.drawImage(frame, 0, 0)
  frame.close()
}
```

### 2. 丢帧策略

```typescript
// 如果解码跟不上，丢弃旧帧
const MAX_QUEUE_SIZE = 5

function decode(data: Uint8Array) {
  if (decoder.decodeQueueSize > MAX_QUEUE_SIZE) {
    console.warn('Dropping frame')
    return
  }
  
  decoder.decode(...)
}
```
