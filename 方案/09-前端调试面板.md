### Step 1: 可停靠面板 ⏳

**文件**：`frontend/src/components/ui/DockPanel.vue`

```vue
<template>
  <div class="dock-panel" :class="[`position-${position}`]" :style="panelStyle">
    <div class="dock-handle" @mousedown="startResize">
      <slot name="header"></slot>
    </div>
    <div class="dock-content">
      <slot></slot>
    </div>
    <div class="dock-resize-handle" @mousedown="startResize"></div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue'

const props = withDefaults(defineProps<{
  position?: 'bottom' | 'right'
  height?: number
  width?: number
}>(), {
  position: 'bottom',
  height: 400,
  width: 500,
})

const emit = defineEmits<{
  (e: 'update:height', value: number): void
  (e: 'update:width', value: number): void
}>()

const currentHeight = ref(props.height)
const currentWidth = ref(props.width)

const panelStyle = computed(() => {
  if (props.position === 'bottom') {
    return { height: `${currentHeight.value}px` }
  } else {
    return { width: `${currentWidth.value}px` }
  }
})

function startResize(e: MouseEvent) {
  e.preventDefault()
  
  const startY = e.clientY
  const startX = e.clientX
  const startHeight = currentHeight.value
  const startWidth = currentWidth.value
  
  function onMouseMove(e: MouseEvent) {
    if (props.position === 'bottom') {
      const delta = startY - e.clientY
      currentHeight.value = Math.max(200, Math.min(800, startHeight + delta))
    } else {
      const delta = startX - e.clientX
      currentWidth.value = Math.max(300, Math.min(800, startWidth + delta))
    }
  }
  
  function onMouseUp() {
    document.removeEventListener('mousemove', onMouseMove)
    document.removeEventListener('mouseup', onMouseUp)
    
    emit('update:height', currentHeight.value)
    emit('update:width', currentWidth.value)
  }
  
  document.addEventListener('mousemove', onMouseMove)
  document.addEventListener('mouseup', onMouseUp)
}
</script>

.dock-panel {
  position: absolute;
  background: #fff;
  border: 1px solid #ddd;
  display: flex;
  flex-direction: column;
  z-index: 100;
}

  bottom: 0;
  left: 0;
  right: 0;
  border-top: 2px solid #1976d2;
}

</style>
```

### Step 2: 增强 DebugPanel ⏳

**文件**：`frontend/src/components/debug/DebugPanel.vue`

```vue
<template>
  <DockPanel
    :position="position"
    v-model:height="panelHeight"
    v-model:width="panelWidth"
  >
    <template #header>
      <div class="debug-header">
        <el-tabs v-model="activeTab" type="card" @tab-change="onTabChange">
          <el-tab-pane label="Logcat" name="logcat">
            <template #label>
              <span>Logcat</span>
              <el-badge :value="logCount" v-if="logCount > 0" class="tab-badge" />
            </template>
          </el-tab-pane>
          <el-tab-pane label="Shell" name="shell" />
          <el-tab-pane label="Network" name="network" />
          <el-tab-pane label="Performance" name="perf" />
          <el-tab-pane label="Apps" name="apps" />
        </el-tabs>

        <div class="debug-actions">
          <el-button size="small" @click="togglePosition">
            {{ position === 'bottom' ? '⇨' : '⇩' }}
          </el-button>
          <el-button size="small" @click="toggleMaximize">
            {{ isMaximized ? '⤓' : '⤢' }}
          </el-button>
          <el-button size="small" @click="close">✕</el-button>
        </div>
      </div>
    </template>

    <component :is="currentComponent" :device-id="deviceId" />
  </DockPanel>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue'
import DockPanel from '@/components/ui/DockPanel.vue'
import LogcatView from './LogcatView.vue'
import ShellView from './ShellView.vue'
import PerfView from './PerfView.vue'
import { useDebugStore } from '@/stores/debug'

const props = defineProps<{ deviceId: string }>()

const debugStore = useDebugStore()

const position = ref<'bottom' | 'right'>('bottom')
const panelHeight = ref(400)
const panelWidth = ref(500)
const isMaximized = ref(false)
const activeTab = ref('logcat')

const logCount = computed(() => debugStore.logs.length)

const currentComponent = computed(() => {
  switch (activeTab.value) {
    case 'logcat': return LogcatView
    case 'shell': return ShellView
    case 'perf': return PerfView
    default: return null
  }
})

function togglePosition() {
  position.value = position.value === 'bottom' ? 'right' : 'bottom'
}

function toggleMaximize() {
  isMaximized.value = !isMaximized.value
  // TODO: 实现最大化逻辑
}

function close() {
  // TODO: 关闭面板
}

function onTabChange(tab: string) {
  // 切换标签时的逻辑
  console.log('Tab changed:', tab)
}

// 快捷键
function handleKeydown(e: KeyboardEvent) {
  // Ctrl+` 切换面板
  if (e.ctrlKey && e.key === '`') {
    e.preventDefault()
    // toggle visibility
  }
  
  // Ctrl+L 聚焦 Logcat
  if (e.ctrlKey && e.key === 'l') {
    e.preventDefault()
    activeTab.value = 'logcat'
  }
  
  // Ctrl+Shift+` 新 Shell
  if (e.ctrlKey && e.shiftKey && e.key === '`') {
    e.preventDefault()
    activeTab.value = 'shell'
  }
}

onMounted(() => {
  window.addEventListener('keydown', handleKeydown)
})

onUnmounted(() => {
  window.removeEventListener('keydown', handleKeydown)
})
</script>
```

### Step 3: 命令面板 ⏳

**文件**：`frontend/src/components/ui/CommandPalette.vue`

```vue
<template>
  <Teleport to="body">
    <div v-if="visible" class="command-palette" @click.self="close">
      <div class="palette-content">
        <input
          ref="inputRef"
          v-model="query"
          type="text"
          placeholder="Type a command..."
          @keydown.escape="close"
          @keydown.enter="executeSelected"
          @keydown.up.prevent="selectPrev"
          @keydown.down.prevent="selectNext"
        />
        
        <div class="command-list">
          <div
            v-for="(cmd, index) in filteredCommands"
            :key="cmd.id"
            :class="['command-item', { selected: index === selectedIndex }]"
            @click="executeCommand(cmd)"
          >
            <div class="command-name">{{ cmd.name }}</div>
            <div class="command-desc">{{ cmd.description }}</div>
            <div class="command-shortcut" v-if="cmd.shortcut">{{ cmd.shortcut }}</div>
          </div>
        </div>
      </div>
    </div>
  </Teleport>
</template>

<script setup lang="ts">
import { ref, computed, onMounted, nextTick } from 'vue'

const props = defineProps<{
  commands: Array<{
    id: string
    name: string
    description: string
    shortcut?: string
    action: () => void
  }>
}>()

const visible = ref(false)
const query = ref('')
const selectedIndex = ref(0)
const inputRef = ref<HTMLInputElement>()

const filteredCommands = computed(() => {
  if (!query.value) return props.commands
  
  const q = query.value.toLowerCase()
  return props.commands.filter(cmd =>
    cmd.name.toLowerCase().includes(q) ||
    cmd.description.toLowerCase().includes(q)
  )
})

function open() {
  visible.value = true
  query.value = ''
  selectedIndex.value = 0
  
  nextTick(() => {
    inputRef.value?.focus()
  })
}

function close() {
  visible.value = false
}

function selectPrev() {
  selectedIndex.value = Math.max(0, selectedIndex.value - 1)
}

function selectNext() {
  selectedIndex.value = Math.min(filteredCommands.value.length - 1, selectedIndex.value + 1)
}

function executeSelected() {
  const cmd = filteredCommands.value[selectedIndex.value]
  if (cmd) {
    executeCommand(cmd)
  }
}

function executeCommand(cmd: any) {
  cmd.action()
  close()
}

// Ctrl+K 打开
onMounted(() => {
  window.addEventListener('keydown', (e) => {
    if (e.ctrlKey && e.key === 'k') {
      e.preventDefault()
      open()
    }
  })
})

defineExpose({ open })
</script>
```
