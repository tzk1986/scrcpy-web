### Step 2: ScrcpyEncoder 实现 ⏳

**文件**：`backend/app/infrastructure/stream/scrcpy.py`

```python
import asyncio
from typing import AsyncIterator
from pathlib import Path

from app.core.config import settings
from app.core.logging import get_logger
from app.domain.ports import EncoderOpts, VideoEncoder

logger = get_logger(__name__)

class ScrcpyEncoder(VideoEncoder):
    """Video encoder using scrcpy-server"""
    
    def __init__(self):
        self.process: asyncio.subprocess.Process | None = None
        self._running = False
        self._device_id: str | None = None
    
    async def start(self, device_id: str, opts: EncoderOpts) -> AsyncIterator[bytes]:
        """Start scrcpy-server and yield H.264 frames"""
        logger.info(
            "starting_scrcpy_encoder",
            device=device_id,
            max_size=opts.max_size,
            bit_rate=opts.bit_rate,
        )
        
        # 推送 scrcpy-server.jar（如果不存在）
        await self._push_server(device_id)
        
        # 启动 scrcpy-server
        self.process = await asyncio.create_subprocess_exec(
            "adb", "-s", device_id, "shell",
            "CLASSPATH=/data/local/tmp/scrcpy-server.jar",
            "com.genymobile.scrcpy.Server", "2.4",
            f"max_size={opts.max_size}",
            f"max_fps={opts.fps}",
            f"video_codec={opts.codec}",
            f"bit_rate={opts.bit_rate}",
            "send_frame_meta=false",
            "raw_stream=true",
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        self._running = True
        self._device_id = device_id
        
        try:
            while self._running:
                if not self.process.stdout:
                    break
                
                # 读取 H.264 帧
                # 注意：实际实现需要处理帧边界
                frame = await self.process.stdout.read(65536)
                if not frame:
                    break
                
                yield frame
        finally:
            await self.stop()
    
    async def stop(self):
        """Stop encoding"""
        self._running = False
        if self.process:
            self.process.kill()
            await self.process.wait()
            self.process = None
        logger.info("scrcpy_encoder_stopped", device=self._device_id)
    
    async def _push_server(self, device_id: str):
        """Push scrcpy-server.jar to device"""
        jar_path = Path(__file__).parent / "scrcpy-server.jar"
        if not jar_path.exists():
            raise FileNotFoundError("scrcpy-server.jar not found")
        
        await asyncio.create_subprocess_exec(
            "adb", "-s", device_id, "push",
            str(jar_path), "/data/local/tmp/scrcpy-server.jar",
        )
```

### Step 3: 帧解析优化 ⏳

**问题**：scrcpy-server 输出的 H.264 流需要正确解析帧边界

**方案**：H.264 NALU 分隔符检测

```python
class H264Parser:
    """Parse H.264 NALU stream"""
    
    def __init__(self):
        self._buffer = bytearray()
    
    def feed(self, data: bytes) -> list[bytes]:
        """Feed data and return complete NALUs"""
        self._buffer.extend(data)
        frames = []
        
        while True:
            # 查找 NALU 起始码 (0x00 0x00 0x00 0x01 或 0x00 0x00 0x01)
            start = self._find_start_code()
            if start == -1:
                break
            
            # 查找下一个 NALU 起始码
            next_start = self._find_start_code(start + 4)
            if next_start == -1:
                # 没有更多帧，等待更多数据
                break
            
            # 提取完整 NALU
            nalu = bytes(self._buffer[start:next_start])
            frames.append(nalu)
            
            # 移除已处理的 NALU
            self._buffer = self._buffer[next_start:]
        
        return frames
    
    def _find_start_code(self, start: int = 0) -> int:
        """Find NALU start code position"""
        for i in range(start, len(self._buffer) - 3):
            if (self._buffer[i] == 0 and 
                self._buffer[i+1] == 0 and 
                self._buffer[i+2] == 1):
                return i
            if (i < len(self._buffer) - 4 and
                self._buffer[i] == 0 and 
                self._buffer[i+1] == 0 and
                self._buffer[i+2] == 0 and
                self._buffer[i+3] == 1):
                return i
        return -1
```

### Step 4: WebSocket 传输 ⏳

**文件**：`backend/app/interfaces/ws/video.py`

```python
from fastapi import APIRouter, WebSocket, WebSocketDisconnect

from app.application.stream_service import StreamService
from app.infrastructure.transport.websocket import WebSocketTransport

router = APIRouter(tags=["video"])

@router.websocket("/ws/video/{device_id}")
async def video_stream(websocket: WebSocket, device_id: str, stream_service: StreamService):
    """Video streaming WebSocket"""
    await websocket.accept()
    transport = WebSocketTransport(websocket)
    
    try:
        # 启动视频流
        async for frame in stream_service.start_stream(device_id):
            await transport.send(frame)
            
            # 同时接收输入事件
            try:
                data = await asyncio.wait_for(websocket.receive_json(), timeout=0.001)
                # TODO: 转发输入到设备
                await handle_input(device_id, data)
            except asyncio.TimeoutError:
                pass
    except WebSocketDisconnect:
        await stream_service.stop_stream(device_id)

async def handle_input(device_id: str, data: dict):
    """Handle input events from client"""
    action = data.get("action")
    
    if action == "touch":
        x, y = data["x"], data["y"]
        await asyncio.create_subprocess_exec(
            "adb", "-s", device_id, "shell", "input", "tap", str(x), str(y)
        )
    elif action == "swipe":
        x1, y1 = data["x1"], data["y1"]
        x2, y2 = data["x2"], data["y2"]
        duration = data.get("duration", 300)
        await asyncio.create_subprocess_exec(
            "adb", "-s", device_id, "shell", "input", "swipe",
            str(x1), str(y1), str(x2), str(y2), str(duration)
        )
    elif action == "key":
        keycode = data["keycode"]
        await asyncio.create_subprocess_exec(
            "adb", "-s", device_id, "shell", "input", "keyevent", str(keycode)
        )
    elif action == "text":
        text = data["text"]
        await asyncio.create_subprocess_exec(
            "adb", "-s", device_id, "shell", "input", "text", text
        )
```

### Step 5: 多设备流管理 ⏳

**问题**：多个客户端同时观看同一设备

**方案**：流共享

```python
class StreamService:
    def __init__(self, encoder: VideoEncoder):
        self.encoder = encoder
        self.active_streams: dict[str, StreamSession] = {}
    
    async def start_stream(self, device_id: str) -> AsyncIterator[bytes]:
        """Start or join video stream"""
        if device_id in self.active_streams:
            # 加入已有流
            session = self.active_streams[device_id]
            async for frame in session.subscribe():
                yield frame
        else:
            # 创建新流
            session = StreamSession(device_id, self.encoder)
            self.active_streams[device_id] = session
            
            try:
                async for frame in session.start():
                    yield frame
            finally:
                del self.active_streams[device_id]

class StreamSession:
    """Shared stream session"""
    
    def __init__(self, device_id: str, encoder: VideoEncoder):
        self.device_id = device_id
        self.encoder = encoder
        self.subscribers: list[asyncio.Queue] = []
    
    async def start(self) -> AsyncIterator[bytes]:
        """Start encoding and broadcast"""
        async for frame in self.encoder.start(self.device_id, EncoderOpts()):
            # 广播给所有订阅者
            for queue in self.subscribers:
                await queue.put(frame)
            yield frame
    
    async def subscribe(self) -> AsyncIterator[bytes]:
        """Subscribe to stream"""
        queue = asyncio.Queue(maxsize=10)
        self.subscribers.append(queue)
        
        try:
            while True:
                frame = await queue.get()
                yield frame
        finally:
            self.subscribers.remove(queue)
```

├── IDR Frame
└── ...
