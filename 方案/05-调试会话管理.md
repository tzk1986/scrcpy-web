#### 1.1 连接池

**问题**：每次查询都创建新连接，性能差

**方案**：使用连接池

```python
import aiosqlite
from contextlib import asynccontextmanager

class SqliteConnectionPool:
    """SQLite connection pool"""
    
    def __init__(self, db_path: str, pool_size: int = 10):
        self.db_path = db_path
        self.pool_size = pool_size
        self._pool: asyncio.Queue[aiosqlite.Connection] = asyncio.Queue(maxsize=pool_size)
        self._initialized = False
    
    async def initialize(self):
        """Initialize connection pool"""
        if self._initialized:
            return
        
        for _ in range(self.pool_size):
            conn = await aiosqlite.connect(self.db_path)
            await self._pool.put(conn)
        
        self._initialized = True
    
    @asynccontextmanager
    async def connection(self):
        """Get connection from pool"""
        conn = await self._pool.get()
        try:
            yield conn
        finally:
            await self._pool.put(conn)
    
    async def close(self):
        """Close all connections"""
        while not self._pool.empty():
            conn = await self._pool.get()
            await conn.close()
```

#### 1.2 批量写入

**问题**：日志逐条写入，性能差

**方案**：批量缓冲写入

```python
class BatchLogWriter:
    """Batch log writer for performance"""
    
    def __init__(self, conn_pool: SqliteConnectionPool, batch_size: int = 100):
        self.conn_pool = conn_pool
        self.batch_size = batch_size
        self._buffer: list[tuple] = []
        self._flush_task: asyncio.Task | None = None
    
    async def start(self):
        """Start background flush task"""
        self._flush_task = asyncio.create_task(self._flush_loop())
    
    async def _flush_loop(self):
        """Flush buffer every 100ms or when full"""
        while True:
            if len(self._buffer) >= self.batch_size:
                await self.flush()
            await asyncio.sleep(0.1)
    
    async def write(self, session_id: str, entry: LogEntry):
        """Write log entry to buffer"""
        self._buffer.append((
            session_id, entry.ts, entry.level,
            entry.pid, entry.tid, entry.tag,
            entry.message, entry.raw
        ))
        
        if len(self._buffer) >= self.batch_size:
            await self.flush()
    
    async def flush(self):
        """Flush buffer to database"""
        if not self._buffer:
            return
        
        batch = self._buffer.copy()
        self._buffer.clear()
        
        async with self.conn_pool.connection() as conn:
            await conn.executemany(
                "INSERT INTO debug_logs VALUES (?,?,?,?,?,?,?,?)",
                batch
            )
            await conn.commit()
    
    async def stop(self):
        """Stop flush task"""
        if self._flush_task:
            self._flush_task.cancel()
        await self.flush()  # Final flush
```

### Step 2: 断线续传机制 ⏳

**问题**：客户端断线重连后丢失日志

**方案**：序列号 + 增量拉取

```python
class DebugService:
    async def _collect_logcat(self, session: DebugSession):
        """Collect logcat with sequence number"""
        seq = 0
        try:
            async for line in self.adb.stream_logcat(session.device_id):
                entry = self._parse_logcat_line(line)
                entry_with_seq = {**entry.__dict__, "seq": seq}
                
                session.log_buffer.append(entry_with_seq)
                if len(session.log_buffer) > self.MAX_LOG_BUFFER:
                    session.log_buffer.pop(0)
                
                await self.batch_writer.write(session.id, entry)
                
                seq += 1
                session.touch()
        except asyncio.CancelledError:
            logger.info("logcat_collection_cancelled", session=session.id)
    
    async def get_logs_since(self, session_id: str, from_seq: int, limit: int = 1000) -> list[dict]:
        """Get logs since sequence number (for reconnection)"""
        session = self.sessions.get(session_id)
        if session:
            # 从内存缓冲获取
            return [
                log for log in session.log_buffer
                if log.get("seq", 0) >= from_seq
            ][:limit]
        else:
            # 从数据库获取（需要添加 seq 字段）
            # TODO: 数据库存储 seq
            return []
```

### Step 3: 会话恢复 ⏳

**问题**：服务重启后会话丢失

**方案**：

```python
class DebugService:
    async def restore_sessions(self):
        """Restore sessions from database on startup"""
        async with self.conn_pool.connection() as conn:
            cursor = await conn.execute(
                "SELECT * FROM debug_sessions WHERE last_active > ?",
                (time.time() - self.SESSION_TTL,)
            )
            rows = await cursor.fetchall()
        
        for row in rows:
            session = DebugSession(
                id=row[0],
                device_id=row[1],
                user_id=row[2],
                created_at=row[3],
                last_active=row[4],
                metadata=json.loads(row[5]) if row[5] else {},
            )
            self.sessions[session.id] = session
            
            # 重新启动 logcat 收集
            self.logcat_tasks[session.id] = asyncio.create_task(
                self._collect_logcat(session)
            )
        
        logger.info("sessions_restored", count=len(rows))
```

### Step 4: 并发控制 ⏳

**问题**：多用户同时操作同一设备

**方案**：会话锁

```python
class DebugService:
    def __init__(self, ...):
        self._session_locks: dict[str, asyncio.Lock] = {}
    
    async def _get_session_lock(self, session_id: str) -> asyncio.Lock:
        """Get or create lock for session"""
        if session_id not in self._session_locks:
            self._session_locks[session_id] = asyncio.Lock()
        return self._session_locks[session_id]
    
    async def exec_shell(self, session_id: str, cmd: str) -> str:
        """Execute shell command with lock"""
        lock = await self._get_session_lock(session_id)
        async with lock:
            session = self.sessions.get(session_id)
            if not session:
                raise ValueError(f"Session not found: {session_id}")
            
            output = await self.adb.shell(session.device_id, cmd)
            await self.repo.save_shell_history(session_id, cmd, output)
            session.shell_history.append(cmd)
            session.touch()
            return output
```

### Step 5: 会话清理 ⏳

**问题**：过期会话占用资源

**方案**：定时清理

```python
class DebugService:
    async def _cleanup_loop(self):
        """Cleanup expired sessions"""
        while True:
            now = time.time()
            expired = [
                sid for sid, session in self.sessions.items()
                if (now - session.last_active) > self.SESSION_TTL
            ]
            
            for session_id in expired:
                await self.close_session(session_id)
                logger.info("session_cleaned", session=session_id)
            
            await asyncio.sleep(3600)  # 每小时检查一次
    
    async def start(self):
        """Start background tasks"""
        self._cleanup_task = asyncio.create_task(self._cleanup_loop())
    
    async def shutdown(self):
        """Shutdown background tasks"""
        if hasattr(self, "_cleanup_task"):
            self._cleanup_task.cancel()
```

```python
        """Append log with memory limit"""
```
