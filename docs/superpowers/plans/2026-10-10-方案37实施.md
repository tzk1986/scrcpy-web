# backend/app/interfaces/http/origin_guard.py
"""Origin/Host 校验中间件（方案 37 A）——CSRF 面：任意网页可向 127.0.0.1:8765 发请求。"""
from urllib.parse import urlparse

from fastapi import Request
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.responses import JSONResponse

from app.core.config import settings


class OriginGuardMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        origin = request.headers.get("origin")
        if origin is not None and not self._allowed(origin, request.headers.get("host")):
            return JSONResponse({"detail": "origin not allowed"}, status_code=403)
        return await call_next(request)

    @staticmethod
    def _allowed(origin: str, host: str | None) -> bool:
        netloc = urlparse(origin).netloc
        return netloc == host or origin in settings().security.cors_origins
```

main.py 在 `app.add_middleware(CORSMiddleware, ...)` 之后加：

"""导出文件名净化（方案 37 A3）：白名单外替换、防穿越/盘符/隐藏文件。"""
import re

_FOLD = re.compile(r"_+")

def sanitize_export_filename(name: str) -> str:
    cleaned = _FOLD.sub("_", "".join(ch if ch.isascii() and (ch.isalnum() or ch in "._-") else "_" for ch in name))

# backend/app/interfaces/ws/protocol.py
"""WS 帧消息解析（方案 37 A4）：非法/缺型消息归一为 None，杜绝 500 与栈泄露。"""
import json

def parse_ws_message(raw: str | bytes) -> dict | None:
    try:
        data = json.loads(raw)
    except (json.JSONDecodeError, TypeError):
        return None
    if not isinstance(data, dict) or not isinstance(data.get("op"), str):
        return None
    return data
```
