```python
def _parse_ls_line(line: str) -> FileEntry | None:
    """解析一行 toybox `ls -la/-ld` 输出为 FileEntry（total 行/空行返回 None）。

    格式（2026-10-10 预研 + 2026-10-12 三机复测）：7 前缀 token
    （权限/links/owner/group/大小/日期/时间）+ 第 8 token 起为名称
    （保留内部空格）；时间列恒 HH:MM（toybox 无年份变体，容忍 HH:MM:SS）。
    软链名称形如 `name -> target`。
    """
    line = line.rstrip("\r\n")
    if not line or line.startswith("total "):
        return None
    parts = line.split(maxsplit=7)
    if len(parts) < 8:
        return None
    permissions, _links, _owner, _group, size, date, time_, rest = parts
    if not permissions or permissions[0] not in "-dl":
        return None
    is_dir = permissions[0] == "d"
    is_symlink = permissions[0] == "l"
    name, link_target = rest, None
    if is_symlink:
        head, sep, tail = rest.rpartition(" -> ")
        if sep:
            name, link_target = head, tail
    mtime = f"{date} {time_}"
    try:
        size_int = int(size)
    except ValueError:
        size_int = 0
    return FileEntry(
        name=name,
        is_dir=is_dir,
        is_symlink=is_symlink,
        size=size_int,
        mtime=mtime,
        permissions=permissions,
        link_target=link_target,
    )
```

5. `AdbCliDriver` 新方法：

```python
    async def list_dir(self, device_id: str, path: str) -> list[FileEntry]:
        """列目录（toybox ls -la）。

        尾斜杠必需：无尾斜杠时 `/sdcard` 这类符号链接只回显链接行本身
        （2026-10-12 实测），补 `/` 才列内容。非 UTF-8 文件名字节以
        errors="replace" 容错（方案 39 §七已知限制）。

        异常：
            AdbError: 路径不存在/无权限（stderr 文本由调用方映射）。
        """
        target = path if path.endswith("/") else path + "/"
        quoted = shlex.quote(target)
        output = await self._run_serial(
            device_id, "shell", f"ls -la {quoted}", decode_errors="replace"
        )
        entries = [_parse_ls_line(line) for line in output.split("\n")]
        return [e for e in entries if e is not None]

    async def stat_size(self, device_id: str, path: str) -> FileEntry:
        """查询单条目（toybox ls -ld，不补尾斜杠——取条目本身）。"""
        quoted = shlex.quote(path)
        output = await self._run_serial(
            device_id, "shell", f"ls -ld {quoted}", decode_errors="replace"
        )
        for line in output.split("\n"):
            entry = _parse_ls_line(line)
            if entry is not None:
                return entry
        raise AdbError(f"stat unparsable for path: {path}")

    async def rename_path(self, device_id: str, old: str, new: str) -> None:
        """同目录重命名（shell mv，参数 shlex.quote）。"""
        await self._run_serial(
            device_id, "shell", f"mv {shlex.quote(old)} {shlex.quote(new)}"
        )

    async def delete_path(self, device_id: str, path: str, is_dir: bool) -> None:
        """删除（目录 rm -r / 文件与软链 rm -f——软链必须 rm -f 不递归目标）。"""
        cmd = "rm -r" if is_dir else "rm -f"
        await self._run_serial(device_id, "shell", f"{cmd} {shlex.quote(path)}")
```

```python
class InvalidPathError(OpenScrcpyException):
    """设备文件路径格式非法（方案 39）。消息不回显原值（防反射面，仿 InvalidIdentifierError）。"""

    def __init__(self, reason: str) -> None:
        super().__init__(f"Invalid path: {reason}", code="INVALID_PATH")


class PathNotFoundError(OpenScrcpyException):
```

```python
"""
文件管理服务
==============

层：应用。

方案 39 核心单元：目录浏览 / 上传 / 下载（多选 zip）/ 删除 / 重命名。

安全：读全盘、写白名单（/sdcard、/data/local/tmp）；路径 normalize 前、
shell quote 后双层防御；上传单文件 512MB；单设备单飞互斥（TransferBusyError）。

暂存：tempfile.TemporaryDirectory（前缀 scrcpy-filejob-），下载清理由
FileResponse background 任务触发 finalize_download；客户端中断可能跳过
background——_sweep 惰性清扫超 15 分钟孤儿暂存目录兜底（已知限制缓解）。

任务表：内存 dict，done/failed 条目 60s TTL 惰性回收。
"""

import asyncio
import posixpath
import shutil
import tempfile
import time
import uuid
import zipfile
from collections.abc import Awaitable, Callable
from pathlib import Path

from app.core.exceptions import (
    AdbError,
    FileTooLargeError,
    InvalidPathError,
    JobNotFoundError,
    NotAFileError,
    PathNotFoundError,
    PermissionDeniedError,
    TargetExistsError,
    TransferBusyError,
)
from app.domain.files import FileEntry, TransferJob
from app.domain.ports import AdbDriver

_DIRNAME_REMOVAL = object()  # 哨兵：delete 预检失败整体 403


class FileService:
    """设备文件管理应用服务（单例：任务表与互斥态进程内共享）。"""

    WRITABLE_ROOTS = ("/sdcard", "/data/local/tmp")
    MAX_UPLOAD_BYTES = 512 * 1024 * 1024
    TRANSFER_TIMEOUT = 900
    JOB_TTL_SECONDS = 60
    STALE_TMPDIR_MINUTES = 15
    CHUNK = 1024 * 1024

    def __init__(self, adb: AdbDriver) -> None:
        self._adb = adb
        self._jobs: dict[str, TransferJob] = {}
        # device_id → 该设备进行中转化的释放事件（互斥即治理，方案 35 语义）
        self._busy: dict[str, asyncio.Event] = {}

    # ---------- 路径净化 ----------

    def normalize_path(self, path: str) -> str:
        """绝对化 + normpath；拒绝相对/空/NUL；消息不回显原值。"""
        if not path or "\x00" in path or not path.startswith("/"):
            raise InvalidPathError("path must be absolute")
        normed = posixpath.normpath(path)
        if "\x00" in normed or not normed.startswith("/"):
            raise InvalidPathError("path normalization failed")
        return normed

    def _ensure_writable(self, path: str) -> None:
        """写目标必须落在白名单前缀内（`root+"/"` 防 /sdcard_x 同头绕过）。"""
        for root in self.WRITABLE_ROOTS:
            if path == root or path.startswith(root + "/"):
                return
        raise PermissionDeniedError(f"写操作仅限白名单目录：{self.WRITABLE_ROOTS}")

    @staticmethod
    def _safe_filename(filename: str) -> str:
        """上传/重命名用文件名净化：取 basename，拒空/./../含 NUL 或 '/'。"""
        name = posixpath.basename(filename or "")
        if not name or name in (".", "..") or "\x00" in name or "/" in name:
            raise InvalidPathError("invalid file name")
        return name

    @staticmethod
    def _map_ls_error(exc: AdbError) -> Exception:
        """设备端 ls 错误文本 → 领域异常（404/403 其余原样）。"""
        msg = str(exc)
        if "No such file or directory" in msg or "Not a directory" in msg:
            return PathNotFoundError("path")  # 调用方带真实 path 重建
        if "Permission denied" in msg:
            return PermissionDeniedError("设备端读取该路径被拒绝")
        return exc

    # ---------- 浏览 ----------

    async def list_dir(self, device_id: str, path: str) -> list[FileEntry]:
        normed = self.normalize_path(path)
        try:
            return await self._adb.list_dir(device_id, normed)
        except AdbError as exc:
            mapped = self._map_ls_error(exc)
            if isinstance(mapped, PathNotFoundError):
                raise PathNotFoundError(normed) from exc
            raise mapped from exc

    async def _stat_or_raise(self, device_id: str, normed: str) -> FileEntry:
        try:
            return await self._adb.stat_size(device_id, normed)
        except AdbError as exc:
            mapped = self._map_ls_error(exc)
            if isinstance(mapped, PathNotFoundError):
                raise PathNotFoundError(normed) from exc
            raise mapped from exc

    # ---------- 互斥 ----------

    def _acquire(self, device_id: str) -> None:
        if device_id in self._busy:
            raise TransferBusyError()
        self._busy[device_id] = asyncio.Event()

    def _release(self, device_id: str) -> None:
        ev = self._busy.pop(device_id, None)
        if ev is not None:
            ev.set()

    # ---------- 下载 ----------

    async def prepare_download(
        self, device_id: str, paths: list[str]
    ) -> tuple[TransferJob, Path, str]:
        """建任务并 pull（多选 zip）至暂存；返回 (job, 本地产物, 下载文件名)。

        锁在成功时由 job 持有——http 层 FileResponse background 调
        finalize_download 释放；异常路径本函数内释放并清理。
        """
        if not paths:
            raise InvalidPathError("empty path list")
        normed = [self.normalize_path(p) for p in paths]
        self._acquire(device_id)
        job = self._new_job(device_id, "download", normed, total_size=0)
        tmp = Path(tempfile.mkdtemp(prefix="scrcpy-filejob-"))
        try:
            if len(normed) == 1:
                entry = await self._stat_or_raise(device_id, normed[0])
                if entry.is_dir:
                    raise NotAFileError(normed[0])
                job.total_size = entry.size
                job.substage = "pulling"
                local = tmp / entry.name
                await self._adb.pull(device_id, normed[0], str(local),
                                     timeout=self.TRANSFER_TIMEOUT)
                filename = entry.name
            else:
                job.substage = "zipping"
                zpath = tmp / "transfer.zip"
                total = 0
                with zipfile.ZipFile(zpath, "w", zipfile.ZIP_DEFLATED) as zf:
                    for p in normed:
                        entry = await self._stat_or_raise(device_id, p)
                        if entry.is_dir:
                            raise NotAFileError(p)
                        total += entry.size
                        raw = tmp / f"raw_{uuid.uuid4().hex}"
                        await self._adb.pull(device_id, p, str(raw),
                                             timeout=self.TRANSFER_TIMEOUT)
                        zf.write(raw, arcname=posixpath.basename(p))
                        raw.unlink()  # 即打包即清（磁盘保留策略）
                job.total_size = total
                job.paths.append  # noqa: B018 —— 不做副作用，见下行
                filename = f"{posixpath.basename(normed[0])}_etc.zip"
            job.substage = "http"
            return job, tmp / filename if len(normed) == 1 else zpath, filename
        except Exception as exc:
            job.status = "failed"
            job.error = str(exc)
            self._finish(job)
            shutil.rmtree(tmp, ignore_errors=True)
            self._release(device_id)
            raise

    def finalize_download(self, job_id: str) -> None:
        """响应发送完成后的清理：删暂存、任务 done、释放设备锁。"""
        job = self._jobs.get(job_id)
        if job is None:
            return
        parent = Path(tempfile.gettempdir())  # 定位：job id 即 tmpdir 后缀
        # 记录于 _jobs 的 workdir 映射在异常路径已 rmtree；成功路径在此清理
        workdir = self._workdirs.pop(job_id, None)
        if workdir is not None:
            shutil.rmtree(workdir, ignore_errors=True)
        _ = parent
        if job.status == "running":
            job.status = "done"
            job.finished_at = time.time()
        self._release(job.device_id)

    # ---------- 上传 ----------

    async def upload(
        self,
        device_id: str,
        dir_path: str,
        filename: str,
        read_chunk: Callable[[int], Awaitable[bytes]],
    ) -> TransferJob:
        """流式接收→暂存→push→即清。read_chunk 由 http 层注入（file.read）。"""
        target_dir = self.normalize_path(dir_path)
        self._ensure_writable(target_dir)
        name = self._safe_filename(filename)
        self._acquire(device_id)
        job = self._new_job(device_id, "upload", [target_dir], total_size=0)
        tmp = Path(tempfile.mkdtemp(prefix="scrcpy-filejob-"))
        self._workdirs[job.id] = tmp
        local = tmp / name
        job.substage = "http"
        written = 0
        try:
            with local.open("wb") as sink:
                while True:
                    chunk = await read_chunk(self.CHUNK)
                    if not chunk:
                        break
                    written += len(chunk)
                    if written > self.MAX_UPLOAD_BYTES:
                        raise FileTooLargeError(self.MAX_UPLOAD_BYTES)
                    sink.write(chunk)
            job.total_size = written
            job.substage = "pushing"
            await self._adb.push(device_id, str(local),
                                 posixpath.join(target_dir, name),
                                 timeout=self.TRANSFER_TIMEOUT)
            self._finish(job)
            return job
        except Exception as exc:
            job.status = "failed"
            job.error = str(exc)
            self._finish(job)
            raise
        finally:
            shutil.rmtree(tmp, ignore_errors=True)
            self._workdirs.pop(job.id, None)
            self._release(device_id)

    # ---------- 重命名 / 删除 ----------

    async def rename(self, device_id: str, path: str, new_name: str) -> None:
        normed = self.normalize_path(path)
        self._ensure_writable(normed)
        name = self._safe_filename(new_name)
        await self._stat_or_raise(device_id, normed)
        target = posixpath.join(posixpath.dirname(normed), name)
        try:
            await self._adb.stat_size(device_id, target)
            raise TargetExistsError(target)
        except AdbError:
            pass  # 目标不存在（ls 非零）——可重命名
        await self._adb.rename_path(device_id, normed, target)

    async def delete(self, device_id: str, paths: list[str]) -> list[dict[str, object]]:
        normed = [self.normalize_path(p) for p in paths]
        for p in normed:  # 预检：任一越界整体 403（spike 判据）
            self._ensure_writable(p)
        results: list[dict[str, object]] = []
        for p in normed:
            try:
                entry = await self._stat_or_raise(device_id, p)
                is_dir = entry.is_dir and not entry.is_symlink
                await self._adb.delete_path(device_id, p, is_dir)
                results.append({"path": p, "ok": True, "error": None})
            except (AdbError, PermissionDeniedError, PathNotFoundError) as exc:
                results.append({"path": p, "ok": False, "error": str(exc)})
        return results

    # ---------- 任务表 ----------

    def get_job(self, device_id: str, job_id: str) -> TransferJob:
        self._sweep()
        job = self._jobs.get(job_id)
        if job is None or job.device_id != device_id:
            raise JobNotFoundError(job_id)
        return job

    def _new_job(self, device_id: str, direction: str, paths: list[str],
                 total_size: int) -> TransferJob:
        self._sweep()
        job = TransferJob(id=uuid.uuid4().hex, device_id=device_id,
                          direction=direction, paths=list(paths),
                          total_size=total_size, created_at=time.time())
        self._jobs[job.id] = job
        self._workdirs.setdefault(job.id, None)  # 由调用方登记实际 workdir
        return job

    def _finish(self, job: TransferJob) -> None:
        if job.status == "running":
            job.finished_at = time.time()

    def _sweep(self) -> None:
        """惰性回收：过期任务条目出表；孤儿暂存目录超时删除。"""
        now = time.time()
        expired = [jid for jid, j in self._jobs.items()
                   if j.status != "running" and j.finished_at
                   and now - j.finished_at > self.JOB_TTL_SECONDS]
        for jid in expired:
            self._jobs.pop(jid, None)
            wd = self._workdirs.pop(jid, None)
            if wd is not None:
                shutil.rmtree(wd, ignore_errors=True)
        cutoff = now - self.STALE_TMPDIR_MINUTES * 60
        root = Path(tempfile.gettempdir())
        try:
            for d in root.glob("scrcpy-filejob-*"):
                try:
                    if d.is_dir() and d.stat().st_mtime < cutoff:
                        shutil.rmtree(d, ignore_errors=True)
                except OSError:
                    continue
        except OSError:
            pass

    # 工作目录登记（成功下载路径由 finalize 清理；upload 即时清理）
    # _workdirs 于 __init__ 声明：
    #     self._workdirs: dict[str, Path | None] = {}
```

实现校正（落码时按此收敛，上面骨架中三处笔误须修正）：
1. `__init__` 需声明 `self._workdirs: dict[str, Path | None] = {}`；
2. 多文件分支 `job.paths.append` 无效行删除；`prepare_download` 成功时须 `self._workdirs[job.id] = tmp`，单文件返回 `tmp / filename` 前先拼 `local`（成功返回 `local, filename`，zip 分支返回 `zpath, filename`——统一 `artifact` 变量）；

```python
"""
设备文件管理 HTTP 端点
========================

层：接口 → HTTP。

    GET  /api/devices/{device_id}/files?path=…          — 目录列表
    GET  /api/devices/{device_id}/file/download?paths=… — 下载（单=原文件，多=zip）
    POST /api/devices/{device_id}/file/upload?dir=…     — multipart 上传（前端逐个单飞）
    POST /api/devices/{device_id}/file/rename           — {path, new_name}
    POST /api/devices/{device_id}/file/delete           — {paths}
    GET  /api/devices/{device_id}/file-jobs/{job_id}    — 任务状态（防跨设备探测）

挂 /api/devices 前缀子路由：多段路径（/files、/file/…）不与单段
/{device_id} 参数路由冲突（devices.py:143 坑仅涉单段字面路径）。
"""

from typing import Any

from fastapi import APIRouter, Depends, File, Query, UploadFile
from fastapi.responses import FileResponse
from pydantic import BaseModel
from starlette.background import BackgroundTask

from app.deps import get_file_service
from app.domain.files import FileEntry, TransferJob
from app.application.file_service import FileService

router = APIRouter(prefix="/api/devices", tags=["files"])


class RenameRequest(BaseModel):
    path: str
    new_name: str


class DeleteRequest(BaseModel):
    paths: list[str]


def _entry_public(e: FileEntry) -> dict[str, Any]: ...  # 7 字段 dict
def _job_public(job: TransferJob) -> dict[str, Any]: ...
    # {id, device_id, direction, paths, total_size, status, substage, error}


@router.get("/{device_id}/files")
async def list_files(device_id: str, service: FileService = Depends(get_file_service),
                     path: str = Query("/sdcard")) -> list[dict[str, Any]]: ...


@router.get("/{device_id}/file/download")
async def download_files(device_id: str, service: FileService = Depends(get_file_service),
                         paths: list[str] = Query(..., min_length=1)) -> FileResponse:
    """下载：单文件原样流、多选 zip。X-File-Job 头回传任务 ID；
    FileResponse background 触发 finalize_download（响应发送完成后清理暂存+释锁）。"""
    job, artifact, filename = await service.prepare_download(device_id, paths)
    return FileResponse(
        artifact,
        filename=filename,
        media_type="application/octet-stream",
        background=BackgroundTask(service.finalize_download, job.id),
        headers={"X-File-Job": job.id},
    )


@router.post("/{device_id}/file/upload")
async def upload_file(device_id: str, file: UploadFile = File(...),
                      service: FileService = Depends(get_file_service),
                      dir: str = Query("/sdcard")) -> dict[str, Any]:
    """项目首个 multipart 端点（python-multipart 已在依赖）。单请求单文件，
    多文件由前端顺序逐个（互斥语义即单飞）。dir 须落在写白名单（403 由服务抛）。"""
    job = await service.upload(device_id, dir, file.filename or "", file.read)
    return _job_public(job)


@router.post("/{device_id}/file/rename")
async def rename_file(device_id: str, body: RenameRequest,
                      service: FileService = Depends(get_file_service)) -> dict[str, bool]:
    await service.rename(device_id, body.path, body.new_name)
    return {"ok": True}


@router.post("/{device_id}/file/delete")
async def delete_files(device_id: str, body: DeleteRequest,
                       service: FileService = Depends(get_file_service)) -> dict[str, Any]:
    return {"results": await service.delete(device_id, body.paths)}


@router.get("/{device_id}/file-jobs/{job_id}")
```

```typescript
// ==================== 文件管理 API（方案 39）====================

export interface FileEntry {
  name: string
  is_dir: boolean
  is_symlink: boolean
  size: number
  mtime: string
  permissions: string
  link_target: string | null
}

export interface FileJob {
  id: string
  device_id: string
  direction: 'download' | 'upload'
  paths: string[]
  total_size: number
  status: 'running' | 'done' | 'failed'
  substage: string | null
  error: string | null
}

export interface DeleteResult {
  path: string
  ok: boolean
  error: string | null
}

/** 设备目录列表。GET /api/devices/{id}/files?path= */
async listFiles(deviceId: string, path: string): Promise<FileEntry[]> {
  const res = await client.get(`/devices/${encodeURIComponent(deviceId)}/files`, {
    params: { path },
  })
  return res.data as FileEntry[]
},

/**
 * 下载文件（单=原文件、多=zip）。paths 手动拼重复查询串
 * （axios 默认序列化为 paths[]=，后端 Query list 需裸重复键）。
 * 超时关闭（大文件）；onDownloadProgress 为 HTTP 段真实百分比。
 * 返回 blob 与任务头 X-File-Job。
 */
async downloadFiles(
  deviceId: string,
  paths: string[],
  onProgress?: (percent: number) => void
): Promise<{ blob: Blob; jobId: string | null }> {
  const q = paths.map((p) => `paths=${encodeURIComponent(p)}`).join('&')
  const res = await client.get<Blob>(
    `/devices/${encodeURIComponent(deviceId)}/file/download?${q}`,
    {
      responseType: 'blob',
      timeout: 0,
      onDownloadProgress: (e) => {
        if (onProgress) {
          const total = e.total ?? 0
          if (total > 0) onProgress(Math.round((e.loaded / total) * 100))
        }
      },
    }
  )
  return { blob: res.data, jobId: res.headers['x-file-job'] as string | undefined ?? null }
},

/** 上传单文件（多文件由调用方顺序循环——设备互斥单飞）。 */
async uploadFile(
  deviceId: string,
  dir: string,
  file: File,
  onProgress?: (percent: number) => void
): Promise<FileJob> {
  const form = new FormData()
  form.append('file', file)
  const res = await client.post<FileJob>(
    `/devices/${encodeURIComponent(deviceId)}/file/upload`,
    form,
    {
      params: { dir },
      timeout: 0,
      onUploadProgress: (e) => {
        if (onProgress) {
          const total = e.total ?? file.size
          if (total > 0) onProgress(Math.round((e.loaded / total) * 100))
        }
      },
    }
  )
  return res.data
},

async renameFile(deviceId: string, path: string, newName: string): Promise<void> {
  await client.post(`/devices/${encodeURIComponent(deviceId)}/file/rename`, {
    path,
    new_name: newName,
  })
},

async deleteFiles(deviceId: string, paths: string[]): Promise<DeleteResult[]> {
  const res = await client.post<{ results: DeleteResult[] }>(
    `/devices/${encodeURIComponent(deviceId)}/file/delete`,
    { paths }
  )
  return res.data.results
},

async getFileJob(deviceId: string, jobId: string): Promise<FileJob> {
  const res = await client.get<FileJob>(
    `/devices/${encodeURIComponent(deviceId)}/file-jobs/${encodeURIComponent(jobId)}`
  )
  return res.data
},
```
